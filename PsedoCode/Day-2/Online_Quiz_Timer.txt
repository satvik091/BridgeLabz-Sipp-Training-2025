CLASS QuizTimer
    PRIVATE:
        timeSlots : ARRAY[INTEGER]   // Circular Array
        totalQuestions : INTEGER
        currentIndex : INTEGER

    PUBLIC:
        CONSTRUCTOR QuizTimer(totalQuestions : INTEGER, timePerQuestion : INTEGER)
            this.totalQuestions = totalQuestions
            timeSlots = NEW ARRAY[totalQuestions]
            FOR i = 0 TO totalQuestions-1
                timeSlots[i] = timePerQuestion
            END FOR
            currentIndex = 0
        END CONSTRUCTOR

        METHOD startTimer()
            FOR q = 0 TO totalQuestions-1
                currentIndex = q MOD totalQuestions
                PRINT "Question " + (q+1) + " : Time allowed = " + timeSlots[currentIndex] + " seconds"
                FOR t = timeSlots[currentIndex] DOWNTO 1
                    IF t == 5 THEN
                        PRINT "⚠️ Warning: Only 5 seconds left!"
                    END IF
                    WAIT 1 second
                END FOR
                PRINT "⏰ Time over for Question " + (q+1)
            END FOR
        END METHOD
END CLASS

FUNCTION mainQuizTimer()
    quiz = NEW QuizTimer(3, 15)   // 3 questions, 15s each
    quiz.startTimer()
END FUNCTION
